Conexión con Google Sheets
🔗 Conectar con Google Sheets — Guía Paso a Paso
⚠️ Google bloquea conexiones directas desde HTML (CORS). La solución es publicar un pequeño script en Google Apps Script. Solo se hace una vez.
1Abrir el Google Sheet de Compensar
2Ir a Extensiones → Apps Script
En el menú superior del Sheet: Extensiones → Apps Script
3Borrar todo y pegar este código
const SHEET_ID = '1YwaTve0UEvlsM8Zy8xYn5UwwqOAfprmPM7orGIj_rYw'; const GID = '992195120'; function doGet(e) { try { var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = getSheetByGid(ss, GID); var data = sheet.getDataRange().getValues(); // Normalizar cabeceras var headers = data[0].map(function(h) { return String(h).trim() .toLowerCase() .replace(/\s+/g, '_') .replace(/[áàä]/g,'a').replace(/[éèë]/g,'e') .replace(/[íìï]/g,'i').replace(/[óòö]/g,'o') .replace(/[úùü]/g,'u').replace(/ñ/g,'n') .replace(/[^a-z0-9_]/g,''); }); var rows = []; for (var i = 1; i < data.length; i++) { var row = {}; for (var j = 0; j < headers.length; j++) { var val = data[i][j]; if (val instanceof Date) { // Detectar si es solo hora (fechas tipo 1899-12-30 son horas en Sheets) var yr = val.getFullYear(); if (yr <= 1900) { // Es un valor de hora var hh = val.getHours().toString().padStart(2,'0'); var mm = val.getMinutes().toString().padStart(2,'0'); val = hh + ':' + mm; } else { val = Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } } row[headers[j]] = val !== undefined && val !== null ? String(val) : ''; } if (Object.values(row).some(function(v){ return v !== ''; })) { rows.push(row); } } var result = ContentService.createTextOutput( JSON.stringify({ ok: true, rows: rows, headers: headers, total: rows.length }) ); result.setMimeType(ContentService.MimeType.JSON); return result; } catch(err) { return ContentService.createTextOutput( JSON.stringify({ ok: false, error: err.message }) ).setMimeType(ContentService.MimeType.JSON); } } function getSheetByGid(ss, gid) { var sheets = ss.getSheets(); for (var i = 0; i < sheets.length; i++) { if (String(sheets[i].getSheetId()) === String(gid)) return sheets[i]; } return ss.getSheets()[0]; }
